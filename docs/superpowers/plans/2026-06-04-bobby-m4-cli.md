// packages/kernel/src/host/kernel-host.ts
import { KernelCommandSchema, type KernelCommand, type KernelEvent } from '@bobby/shared';
import { Orchestrator } from '../brain/orchestrator';
import type { ModelClient } from '../model/model-client';

type Listener = (e: KernelEvent) => void;
type GateDecision = 'allow' | 'deny';

export class KernelHost {
  private listeners = new Set<Listener>();
  private pendingGates = new Map<string, (d: GateDecision) => void>();
  constructor(private makeModel: () => ModelClient) {}

  subscribe(fn: Listener) { this.listeners.add(fn); return () => this.listeners.delete(fn); }
  private emit(e: KernelEvent) { for (const fn of this.listeners) fn(e); }

  async send(cmd: KernelCommand): Promise<void> {
    const c = KernelCommandSchema.parse(cmd); // 守门
    switch (c.type) {
      case 'startTask': {
        const orch = new Orchestrator(this.makeModel());
        orch.on((e) => this.emit(e));
        await orch.startTask(c.input);
        return;
      }
      case 'approveGate': {
        const resolve = this.pendingGates.get(c.gateId);
        if (resolve) { this.pendingGates.delete(c.gateId); resolve(c.decision); }
        return;
      }
      default: return; // answer/abort/getTrace 在 M7 配置/会话里补全
    }
  }

  /** 供编排器在危险动作处调用：发 gate_request，挂起直到 approveGate。 */
  requestGate(taskId: string, gateId: string, reason: string): Promise<GateDecision> {
    this.emit({ type: 'gate_request', taskId, gateId, reason });
    return new Promise((res) => this.pendingGates.set(gateId, res));
  }
}
```

// packages/cli/src/ui/view-model.ts
import type { KernelEvent } from '@bobby/shared';
export interface VM { lines: string[]; status: 'running' | 'done' | 'failed' | 'blocked'; pendingGate?: { gateId: string; reason: string }; }
export const initialVM = (): VM => ({ lines: [], status: 'running' });
export function reduceEvent(vm: VM, e: KernelEvent): VM {
  switch (e.type) {
    case 'intent_proposed': return { ...vm, lines: [...vm.lines, `🎯 意图：${e.contract.goal}`] };
    case 'plan_ready': return { ...vm, lines: [...vm.lines, `🗺️ 计划：${e.steps.length} 步`] };
    case 'step_started': return { ...vm, lines: [...vm.lines, `▶️ ${e.stepId}`] };
    case 'tool_called': return { ...vm, lines: [...vm.lines, `🔧 ${e.tool}`] };
    case 'verdict': return { ...vm, lines: [...vm.lines, `⚖️ ${e.verdict.acId}: ${e.verdict.result}`] };
    case 'gate_request': return { ...vm, pendingGate: { gateId: e.gateId, reason: e.reason } };
    case 'final_result': return { ...vm, status: e.status, lines: [...vm.lines, `🏁 ${e.status}`] };
    case 'error': return { ...vm, lines: [...vm.lines, `❌ ${e.message}`] };
    default: return vm;
  }
}
```

// packages/cli/src/headless.ts
import type { KernelHost } from '@bobby/kernel';
export async function runHeadless(host: KernelHost, input: string): Promise<{ exitCode: number; status: string }> {
  let status = 'failed';
  host.subscribe((e) => { if (e.type === 'final_result') status = e.status; });
  await host.send({ type: 'startTask', input });
  return { exitCode: status === 'done' ? 0 : 1, status }; // 不撒谎：非 done 一律非 0
}
```

// packages/cli/src/ui/App.tsx
import React from 'react';
import { Box, Text, useInput } from 'ink';
import type { VM } from './view-model';

export function App({ vm, onSubmit, onGate }: { vm: VM; onSubmit: (s: string) => void; onGate: (d: 'allow' | 'deny') => void }) {
  useInput((input, key) => {
    if (vm.pendingGate) { if (input === 'y') onGate('allow'); if (input === 'n') onGate('deny'); return; }
    if (key.return) onSubmit('\n'); // 真实输入框在 index.tsx 用 ink TextInput 接管
  });
  return (
    <Box flexDirection="column">
      {vm.lines.map((l, i) => <Text key={i}>{l}</Text>)}
      {vm.pendingGate && <Text color="yellow">⚠️ {vm.pendingGate.reason}  允许? [y/n]</Text>}
      {!vm.pendingGate && <Text color="gray">状态: {vm.status}</Text>}
    </Box>
  );
}
```
