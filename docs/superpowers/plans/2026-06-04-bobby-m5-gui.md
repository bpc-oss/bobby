```ts
// packages/gui/electron/main.ts
import { app, BrowserWindow, ipcMain } from 'electron';
import { KernelHost } from '@bobby/kernel';
let host: KernelHost;
app.whenReady().then(() => {
  const makeModel = () => { throw new Error('未配置 DeepSeek Key（见设置/首启向导）'); }; // M6 替换为真实客户端
  host = new KernelHost(makeModel);
  const win = new BrowserWindow({ width: 1100, height: 760, webPreferences: { preload: `${__dirname}/preload.js`, contextIsolation: true } });
  host.subscribe((e) => win.webContents.send('kernel:event', e));
  ipcMain.handle('kernel:command', (_e, cmd) => host.send(cmd));
  win.loadURL(process.env.VITE_DEV_SERVER_URL ?? `file://${__dirname}/../dist/index.html`);
});
```
```ts
// packages/gui/electron/preload.ts
import { contextBridge, ipcRenderer } from 'electron';
contextBridge.exposeInMainWorld('bobby', {
  send: (cmd: unknown) => ipcRenderer.invoke('kernel:command', cmd),
  onEvent: (cb: (e: unknown) => void) => ipcRenderer.on('kernel:event', (_e, ev) => cb(ev)),
});
```

```ts
// packages/gui/src/ipc/contract.ts
import type { KernelEvent } from '@bobby/shared';
interface Bridge { send: (cmd: unknown) => Promise<void>; onEvent: (cb: (e: KernelEvent) => void) => void; }
declare global { interface Window { bobby: Bridge; } }
export function makeKernelClient() {
  return {
    startTask: (input: string) => window.bobby.send({ type: 'startTask', input }),
    approveGate: (gateId: string, decision: 'allow' | 'deny') => window.bobby.send({ type: 'approveGate', gateId, decision }),
    onEvent: (cb: (e: KernelEvent) => void) => window.bobby.onEvent(cb),
  };
}
```

```ts
// packages/gui/src/lib/evidence-format.ts
import type { Evidence } from '@bobby/shared';
export interface PlainEvidence { summary: string; ok: boolean; detail: string; }
export function toPlainLanguage(ev: Evidence): PlainEvidence {
  const p = ev.payload as Record<string, unknown>;
  switch (ev.evidenceType) {
    case 'command_output': {
      const ok = p.exitCode === 0;
      return { summary: `运行了命令（${ok ? '成功' : '失败'}）`, ok, detail: String(p.stdout ?? p.stderr ?? '') };
    }
    case 'file_exists': return { summary: p.exists ? '确认文件已生成' : '文件没生成', ok: p.exists === true, detail: String(p.path ?? '') };
    case 'file_diff': return { summary: '修改了文件', ok: true, detail: String(p.path ?? '') };
    case 'pro_review': return { summary: 'Pro 复核', ok: true, detail: JSON.stringify(p) };
    case 'quote_with_location': return { summary: '逐项核对留痕', ok: true, detail: JSON.stringify(p) };
    default: return { summary: ev.evidenceType, ok: true, detail: JSON.stringify(p) };
  }
}
```

```ts
// packages/gui/src/store/app-store.ts
import type { Evidence, KernelEvent } from '@bobby/shared';
export interface AppState { steps: string[]; evidence: Evidence[]; status: string; pendingGate?: { gateId: string; reason: string }; }
export const initial = (): AppState => ({ steps: [], evidence: [], status: 'running' });
export function reduce(s: AppState, e: KernelEvent): AppState {
  switch (e.type) {
    case 'step_started': return { ...s, steps: [...s.steps, e.stepId] };
    case 'evidence_produced': return { ...s, evidence: [...s.evidence, e.evidence] };
    case 'gate_request': return { ...s, pendingGate: { gateId: e.gateId, reason: e.reason } };
    case 'final_result': return { ...s, status: e.status, pendingGate: undefined };
    default: return s;
  }
}
```
